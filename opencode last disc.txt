    public partial class MainWindow : Window
    {
        private bool _isLoading = true;
        
        public MainWindow()
        {
            InitializeComponent();
            
            // Make window draggable from anywhere (borderless style)
            MouseLeftButtonDown += Window_MouseLeftButtonDown;
            
            // Ensure the dashboard is properly initialized on startup
            Loaded += MainWindow_Loaded;
        }

        private void MainWindow_Loaded(object sender, RoutedEventArgs e)
        {
            // Start the loading animation when window is loaded
            // Use a simple approach to avoid timing issues
            Task.Run(() => StartLoadingAnimation());
        }

        private async void StartLoadingAnimation()
        {
            try
            {
                _isLoading = true;
                
                // Simulate loading for 1 second (100 steps at 10ms each)
                for (int i = 0; i < 100; i++)
                {
                    await Task.Delay(10);
                    
                    if (!_isLoading) break;
                }
                
                _isLoading = false;
                
                // Ensure the main view is displayed
                Dispatcher.Invoke(() =>
                {
                    var viewModel = DataContext as MainViewModel;
                    if (viewModel != null)
                    {
                        viewModel.CurrentView = DashboardView.Main;
                    }
                });
            }
            catch (Exception ex)
            {
                // Log error but don't crash the application
                System.Diagnostics.Debug.WriteLine($"Loading error: {ex.Message}");
            }
        }

        // Window control handlers - close button minimizes to tray instead of closing
